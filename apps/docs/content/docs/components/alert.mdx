## Usage

```tsx
import { Alert, AlertDescription, AlertTitle } from "@moe/registry/ui/alert";
import { AlertCircle } from "lucide-react-native";

return (
  <Alert icon={AlertCircle}>
    <AlertTitle>Heads up!</AlertTitle>
    <AlertDescription>
      You can add components to your app using the cli.
    </AlertDescription>
  </Alert>
);
```

## Props

| Prop            | Type                         | Default     | Description                   |
| --------------- | ---------------------------- | ----------- | ----------------------------- |
| `icon`          | `LucideIcon`                 | -           | Icon to display in the alert  |
| `variant`       | `'default' \| 'destructive'` | `'default'` | The visual style of the alert |
| `iconClassName` | `string`                     | -           | Additional classes for icon   |
